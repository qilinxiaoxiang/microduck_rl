7888:     upper_sensor: str = "object_upper_contact",
7889:     floor_sensor: str = "object_floor_contact",
7890: ) -> dict:
7891:     """Per-step grasp booleans, cached on the step counter (several reward terms read them)."""
7892:     c = _gg_cache(env)
7893:     key = (env.common_step_counter, asset_name)
7894:     if c.get("grasp_state_key") == key:
7895:         return c["grasp_state"]

7905:               held=pinched & ~floor & (lift > 0.01))
7906:     c["grasp_state_key"] = key
7907:     c["grasp_state"] = st
